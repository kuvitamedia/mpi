Pseudocode: Validasi Login Sistem

WHILE & IF-ELSE 
ALGORITMA Validasi_Login_Sistem
DEKLARASI: username_input, password_input : STRING
           percobaan : INTEGER = 0

ALGORITMA:
  WHILE (percobaan < 3) DO
    READ username_input, password_input
    IF (username_input == 'siswa' AND password_input == '12345') THEN
      PRINT 'Akses Diterima! Selamat Belajar.'
      EXIT_PROGRAM
    ELSE
      percobaan = percobaan + 1
      PRINT 'Login Gagal. Sisa Percobaan: ' + (3 - percobaan)
    ENDIF
  ENDWHILE
  PRINT 'Akun Terkunci! Hubungi Administrator.'
